cmake_minimum_required(VERSION 3.16)
project(dlslime_host_region_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(DLSLIME_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(host_region_test host_region_test.cpp
    ${DLSLIME_ROOT}/dlslime/csrc/engine/region/host_region.cpp)
target_include_directories(host_region_test PRIVATE ${DLSLIME_ROOT})
find_package(Threads REQUIRED)
target_link_libraries(host_region_test PRIVATE Threads::Threads)
option(REGION_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(REGION_SANITIZE)
    target_compile_options(host_region_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(host_region_test PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME host_region_test COMMAND host_region_test)
set_tests_properties(host_region_test PROPERTIES TIMEOUT 60)
