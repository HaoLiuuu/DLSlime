cmake_minimum_required(VERSION 3.16)
project(dlslime_tcp_dram_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(DLSLIME_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${DLSLIME_ROOT}/cmake/StandaloneAsio.cmake")
set(TCP_ROOT "${DLSLIME_ROOT}/dlslime/csrc/engine/tcp")
add_library(tcp_benchmark_transport STATIC
    ${TCP_ROOT}/tcp_context.cpp ${TCP_ROOT}/tcp_connection_pool.cpp
    ${TCP_ROOT}/tcp_memory_pool.cpp ${TCP_ROOT}/tcp_session.cpp ${TCP_ROOT}/tcp_endpoint.cpp)
target_include_directories(tcp_benchmark_transport PUBLIC ${DLSLIME_ROOT})
target_link_libraries(tcp_benchmark_transport PUBLIC asio::asio)
option(TCP_SANITIZE "Sanitize correctness checks, never performance measurements" OFF)
if(TCP_SANITIZE)
    target_compile_options(tcp_benchmark_transport PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(tcp_benchmark_transport PUBLIC -fsanitize=address,undefined)
endif()
add_executable(tcp_dram_bench tcp_dram_bench.cpp)
target_link_libraries(tcp_dram_bench PRIVATE tcp_benchmark_transport)
